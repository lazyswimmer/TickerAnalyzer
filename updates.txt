def validate_ticker_exists(ticker: str) -> Dict[str, Any]:
    """
    Fast guard for the web API.

    It catches obvious bad input before running the expensive full assessment,
    then asks Yahoo Finance for recent trading history. For this MVP, a ticker
    is treated as valid only if Yahoo returns recent price rows.
    """
    ticker = str(ticker or "").upper().strip()

    if not ticker:
        return {
            "exists": False,
            "error": "Please enter a ticker."
        }

    if not re.fullmatch(r"[A-Z0-9][A-Z0-9.\-]{0,14}", ticker):
        return {
            "exists": False,
            "error": "Ticker format looks invalid. Use symbols like AAPL, MSFT, BRK-B, or BRK.B.",
        }

    try:
        stock = yf.Ticker(ticker)
        recent_history = stock.history(period="1mo", auto_adjust=False)

        if recent_history is None or recent_history.empty:
            return {
                "exists": False,
                "error": f"I could not find recent market data for '{ticker}'. Please check the ticker symbol.",
            }

        return {
            "exists": True,
            "ticker": ticker
        }

    except Exception as exc:
        return {
            "exists": False,
            "error": f"Could not validate ticker '{ticker}': {exc}",
        }

def plain_score_label(score):
    """Simple user-facing score label for the story report."""
    if score is None:
        return "Unknown"

    try:
        score = float(score)
    except Exception:
        return "Unknown"

    if score >= 7.5:
        return "Strong"
    if score >= 6:
        return "Healthy"
    if score >= 4.5:
        return "Mixed"

    return "Weak"


def split_strengths_and_risks(summary):
    """
    Turn category scores and notes into a small set of human-readable strengths
    and risks. This keeps the UI focused on the few signals that matter most.
    """
    scores = summary.get("category_scores", {}) or {}
    notes = summary.get("notes", {}) or {}

    strengths = []
    risks = []

    for category, score in scores.items():
        category_notes = notes.get(category, []) or []
        first_note = category_notes[0] if category_notes else "Available data is limited."

        if isinstance(score, (int, float)) and score >= 6.5:
            strengths.append({
                "title": category,
                "signal": plain_score_label(score),
                "score": score,
                "explanation": first_note,
            })

        elif isinstance(score, (int, float)) and score <= 5.0:
            risks.append({
                "title": category,
                "signal": plain_score_label(score),
                "score": score,
                "explanation": first_note,
            })

    strengths = sorted(
        strengths,
        key=lambda x: x.get("score", 0),
        reverse=True
    )[:3]

    risks = sorted(
        risks,
        key=lambda x: x.get("score", 10)
    )[:3]

    if not risks:
        risks.append({
            "title": "No single category dominates the risk picture",
            "signal": "Watch",
            "score": None,
            "explanation": "The screen did not find one obvious weak area, but this should still be checked against filings, earnings updates, and company-specific news.",
        })

    if not strengths:
        strengths.append({
            "title": "No clear standout strength",
            "signal": "Mixed",
            "score": None,
            "explanation": "The available data does not show a category that clearly offsets the main risks.",
        })

    return strengths, risks
def build_highlighted_signals(summary):
    """Create a compact signal strip for the top of the report."""
    scores = summary.get("category_scores", {}) or {}

    numeric_scores = {
        k: v for k, v in scores.items()
        if isinstance(v, (int, float))
    }

    strongest = max(numeric_scores, key=numeric_scores.get) if numeric_scores else None
    weakest = min(numeric_scores, key=numeric_scores.get) if numeric_scores else None

    signals = [
        {
            "label": "Risk level",
            "value": summary.get("risk_label", "Unknown"),
            "tone": "risk",
            "description": f"General risk score: {summary.get('generalized_risk_score', 'N/A')}/10",
        },
        {
            "label": "Overall profile",
            "value": plain_score_label(summary.get("overall_quality_score")),
            "tone": "quality",
            "description": f"Quality score: {summary.get('overall_quality_score', 'N/A')}/10",
        },
    ]

    if strongest:
        signals.append({
            "label": "Best signal",
            "value": strongest,
            "tone": "positive",
            "description": f"{plain_score_label(scores.get(strongest))}: {scores.get(strongest)}/10",
        })

    if weakest:
        signals.append({
            "label": "Main watch area",
            "value": weakest,
            "tone": "warning",
            "description": f"{plain_score_label(scores.get(weakest))}: {scores.get(weakest)}/10",
        })

    return signals
def build_simple_metric_snapshot(summary):
    """
    Keep only a short, plain-English financial health overview.
    The detailed metric table is still returned for future use, but the main UI
    should lead with this smaller set.
    """
    metrics = summary.get("metric_snapshot", {}) or {}

    keep = [
        "Company",
        "Sector",
        "Market Cap",
        "Revenue Growth",
        "Operating Margin",
        "FCF Margin",
        "Net Debt / EBITDA",
        "Interest Coverage",
        "Current Drawdown",
    ]

    return {
        k: metrics.get(k)
        for k in keep
        if metrics.get(k) not in (None, "", "N/A")
    }


def build_financial_health_overview(summary):
    scores = summary.get("category_scores", {}) or {}
    metrics = summary.get("metric_snapshot", {}) or {}

    balance = plain_score_label(scores.get("Balance Sheet")).lower()
    earnings = plain_score_label(scores.get("Earnings Quality")).lower()
    growth = plain_score_label(scores.get("Growth / Profitability")).lower()

    parts = [
        f"Balance sheet: {balance}.",
        f"Earnings quality: {earnings}.",
        f"Growth and profitability: {growth}.",
    ]

    metric_clues = []

    for label in [
        "Revenue Growth",
        "Operating Margin",
        "FCF Margin",
        "Net Debt / EBITDA",
    ]:
        value = metrics.get(label)

        if value not in (None, "", "N/A"):
            metric_clues.append(f"{label}: {value}")

    return {
        "summary": " ".join(parts),
        "supporting_points": metric_clues[:4],
    }

function getApiBaseUrl() {
    // Best case: the page is served by Flask at http://127.0.0.1:5000/.
    // Fallback: if opened as a file or from a static dev server, call Flask directly.
    if (window.location.protocol === "file:" || window.location.port !== "5000") {
        return "http://127.0.0.1:5000";
    }

    return "";
}


function explainFetchFailure(error) {
    return (
        "Could not reach the Flask API. " +
        "Make sure the server is running with `python app.py`, then open `http://127.0.0.1:5000/`. " +
        "If you are using VS Code Live Server or opening index.html directly, the page may be running from the wrong address. " +
        "Browser error: " + error
    );
}

async function calculateRisk() {
    const stock = document.getElementById("stockInput").value.toUpperCase().trim();

    const status = document.getElementById("status");
    const output = document.getElementById("output");

    if (!stock) {
        status.innerHTML = '<p class="error">Please enter a stock ticker.</p>';
        output.style.display = "none";
        return;
    }

    status.innerHTML = '<p class="loading">Building a plain-English report for ' + stock + '...</p>';
    output.style.display = "none";

    try {
        const apiBaseUrl = getApiBaseUrl();

        const response = await fetch(
            apiBaseUrl + "/api/assessment?ticker=" + encodeURIComponent(stock)
        );

        const rawResponse = await response.text();

        let data;

        try {
            data = JSON.parse(rawResponse);
        } catch (parseError) {
            throw new Error(
                "The server returned HTML instead of JSON. Make sure you are running the Flask app with `python app.py` and opening http://127.0.0.1:5000/. Server response starts with: " +
                rawResponse.slice(0, 80)
            );
        }

        if (!response.ok || !data.success) {
            status.innerHTML = '<p class="error">' + escapeHtml(data.error || "Assessment failed.") + '</p>';
            return;
        }

        renderAssessment(data);

        status.innerHTML = "";
        output.style.display = "block";

    } catch (error) {
        status.innerHTML = '<p class="error">' + escapeHtml(explainFetchFailure(error)) + '</p>';
    }
}

function escapeHtml(value) {
    return String(value)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
}
